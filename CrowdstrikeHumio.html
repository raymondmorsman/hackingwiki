Email timelines:

#type match {

"o365-messagetrace" => email.from = ?User

| email.to != "*@corbion.com"

| email.subject != "Automatic reply:*"

| user := email.from | operation := event.application | event := email.subject | argument1 := email.to;

"o365-audit" => source.user = ?User

| operation := microsoft365.operation | user:= source.user | microsoft365.operation match {

"HardDelete" => event := email.subject;

"SoftDelete" => event := email.subject;

"Move" => event := email.subject | argument1 := microsoft365.destinationFolderPath;

"MoveToDeletedItems" => event := email.subject;

* => event := microsoft365.objectId;}

} | select([@timestamp, operation, event, argument1])
