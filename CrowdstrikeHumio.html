Statistics on AI

#event_simpleName=DnsRequest

| join(query={#event_simpleName=ProcessRollup2}, field=source.host, include=[source.user], mode=inner)


 

| UserName := source.user

| ComputerName := source.host


 

| !regex(field=UserName, "^(SYSTEM|LOCAL SERVICE|.*\\$)$")

| regex(field=ComputerName, "^(?<Region>..)")

// | !in(field=Region, values=["CE","OT","PT","DE","KI","PA"])


 

| AI_Flag := if(match(file="generative-ai-domains.csv", field=[domain], column=domain, ignoreCase=true, mode=glob), then=1, else=0)

| AI_User := if(AI_Flag > 0, then=UserName, else=null)


 

| groupBy([Region], function=[count(UserName, distinct=true, as=ActiveUsers), sum(AI_Flag, as=AIRequests), count(AI_User, distinct=true, as=AIUser), count(as=TotalRequests)])

| AI_Req_Percentage := ((AIRequests / TotalRequests) * 100)

| AI_Req_Percentage := format("%.2f", field=[AI_Req_Percentage])


 

| AI_User_Perc := ((AIUser / ActiveUsers) * 100)

| AI_User_Perc := format("%.2f", field=[AI_User_Perc])

| sort(AI_User_Perc, order=desc)
